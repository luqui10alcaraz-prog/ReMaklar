#!/bin/sh
set -eu
p=/var/tmp/graphics011-stage102
out=/var/tmp/graphics011-capacity102-results
exe=/var/tmp/graphics011-build101/qemu-ppc
test ! -e "$out";mkdir "$out";exec > "$out/session.txt" 2>&1
set -x
trap 'status=$?;echo script_exit=$status > "$out/script-status.txt";sync;mount -t msdos /dev/disk2 /exchange;cd /var/tmp;paths=graphics011-capacity102-results;if test -d graphics011-joint102;then paths="$paths graphics011-joint102";fi;gnutar czf /exchange/RUNS/cap102.tgz $paths;sync;umount /exchange' 0
probe=/var/tmp/graphics009-r8/run_graphics_008.pl
for arch in i386 ppc;do gcc2 -no-cpp-precomp -arch "$arch" "$p/capacity.c" -o "$out/capacity-$arch" > "$out/capacity-$arch-build.txt" 2>&1;done
export REMAKLAR_QUIET_TLS=1
perl "$probe" "$out/capacity-native" "$out/capacity-i386"
perl "$probe" "$out/capacity-previous" /var/tmp/graphics011-build100/qemu-ppc --root /var/tmp/jaguar "$out/capacity-ppc"
grep -q '^leader_exit=125$' "$out/capacity-previous.status"
perl "$probe" "$out/capacity-jaguar" "$exe" --root /var/tmp/jaguar "$out/capacity-ppc"
cmp "$out/capacity-native.stdout" "$out/capacity-jaguar.stdout"
for mode in native jaguar;do
 grep -q '^leader_exit=0$' "$out/capacity-$mode.status"
 grep -q '^timeout=0$' "$out/capacity-$mode.status"
done
gcc2 -no-cpp-precomp -arch ppc "$p/stress.c" -o "$out/names-stress-ppc" > "$out/names-stress-build.txt" 2>&1
perl "$probe" "$out/names-stress" "$exe" --root /var/tmp/jaguar "$out/names-stress-ppc"
grep -q '^leader_exit=0$' "$out/names-stress.status"
grep -q '^timeout=0$' "$out/names-stress.status"
echo probes_completed > "$out/phase.txt"
export REMAKLAR_JOINT_SAMPLES=72
export REMAKLAR_OBSERVE_APPKIT=1
export REMAKLAR_OBSERVE_APPKIT_OFFSETS=624e0,1c7758,1c779c,1c7800
export REMAKLAR_TRACE_PATHS=1
perl "$p/joint.pl" /var/tmp/graphics011-joint102 "$exe" /var/tmp/jaguar
