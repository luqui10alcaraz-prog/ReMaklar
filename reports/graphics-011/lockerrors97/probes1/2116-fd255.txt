#!/bin/sh
set -eu
p=/var/tmp/graphics011-stage97
out=/var/tmp/graphics011-filelock97-results
work=/var/tmp/graphics011-build97
test ! -e "$out";test ! -e "$work";mkdir "$out" "$work";exec > "$out/session.txt" 2>&1
set -x
df -k > "$out/disk.txt"
trap 'status=$?;echo script_exit=$status > "$out/script-status.txt";sync;mount -t msdos /dev/disk2 /exchange;cd /var/tmp;paths=graphics011-filelock97-results;if test -d graphics011-joint97;then paths="$paths graphics011-joint97";fi;gnutar czf /exchange/flck97.tgz $paths;sync;umount /exchange' 0
cp "$p/darwin-kernel.c" /var/tmp/graphics011-run1/source/darwin-user/remaklar/darwin-kernel.c
make -C /var/tmp/graphics011-build81/build SRC_PATH=/var/tmp/graphics011-run1/source subdir-ppc-darwin-user > "$out/build.txt" 2>&1
cp /var/tmp/graphics011-build81/build/ppc-darwin-user/qemu-ppc "$work/qemu-ppc"
exe="$work/qemu-ppc"
probe=/var/tmp/graphics009-r8/run_graphics_008.pl
for arch in i386 ppc;do gcc2 -no-cpp-precomp -arch "$arch" "$p/filelock.c" -o "$out/filelock-$arch" > "$out/filelock-$arch-build.txt" 2>&1;done
export REMAKLAR_QUIET_TLS=1
perl "$probe" "$out/filelock-native" "$out/filelock-i386"
perl "$probe" "$out/filelock-previous" /var/tmp/graphics011-build96/qemu-ppc --root /var/tmp/jaguar "$out/filelock-ppc"
grep -q '^leader_exit=1$' "$out/filelock-previous.status"
perl "$probe" "$out/filelock-jaguar" "$exe" --root /var/tmp/jaguar "$out/filelock-ppc"
cmp "$out/filelock-native.stdout" "$out/filelock-jaguar.stdout"
for mode in native jaguar;do
 grep -q '^leader_exit=0$' "$out/filelock-$mode.status"
 grep -q '^timeout=0$' "$out/filelock-$mode.status"
done
echo probes_completed > "$out/phase.txt"
export REMAKLAR_OBSERVE_APPKIT=1
export REMAKLAR_OBSERVE_APPKIT_OFFSETS=624e0,1c7758,1c779c,1c7800
export REMAKLAR_TRACE_PATHS=1
perl "$p/joint.pl" /var/tmp/graphics011-joint97 "$exe" /var/tmp/jaguar
